SETUP INSTRUCTIONS:
1. Create .github/workflows/snake.yml in this repository.
2. Add the workflow below.
3. Commit and push the workflow to GitHub.
4. In repository Settings > Actions > General, enable the required workflow permissions.
5. Run the workflow once. It creates the output branch and SVG files.

.github/workflows/snake.yml:

name: Generate Contribution Snake

on:
  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 5

    steps:
      - name: Generate contribution snake
        uses: Platane/snk@v3
        with:
          github_user_name: abhinav-swe
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark

      - name: Publish SVGs to output branch
        uses: crazy-max/ghaction-github-pages@v4
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
